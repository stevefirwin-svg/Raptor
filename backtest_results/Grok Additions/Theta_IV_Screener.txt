@staticmethod
def d1(S, K, T, r, sigma):

@staticmethod

d_2 = BlackScholes.d2(S, K, T, r, sigma)

d2 = BlackScholes.d2(S, K, T, r, sigma)

class BlackScholes:
    @staticmethod
    def greeks(S, K, T, r, sigma, opt_type="call"):
        d1 = (np.log(S/K) + (r + 0.5*sigma**2)*T) / (sigma * np.sqrt(T))
        d2 = d1 - sigma * np.sqrt(T)
        nd1 = norm.cdf(d1)
        nd2 = norm.cdf(d2)
        n_d1 = norm.pdf(d1)
        if opt_type == "call":
            delta = nd1
            theta = -(S * n_d1 * sigma)/(2*np.sqrt(T)) - r*K*np.exp(-r*T)*nd2
            vega = S * n_d1 * np.sqrt(T)
        else:
            delta = nd1 - 1
            theta = -(S * n_d1 * sigma)/(2*np.sqrt(T)) + r*K*np.exp(-r*T)*(1 - nd2)
            vega = S * n_d1 * np.sqrt(T)
        return {"delta": delta, "theta": theta, "vega": vega}

    @staticmethod
    def iv(market_price, S, K, T, r, opt_type):
        def obj(sigma):
            price = BlackScholes.greeks(S, K, T, r, sigma, opt_type)["price"] if "price" in BlackScholes.greeks(S, K, T, r, sigma, opt_type) else 0
            return price - market_price
        try:
            return brentq(obj, 0.01, 5.0)
        except:
            return np.nan
